import { runNodeProductionRender } from "./renderRuntime";

interface IActionResult {
  error: string;
  out: string;
  status: number;
}

const actions = JSON.parse(process.argv[2] ?? "[]") as string[][];
const results: IActionResult[] = [];
for (const argv of actions) {
  const printed: string[] = [];
  const write = process.stdout.write.bind(process.stdout);
  process.stdout.write = ((chunk: unknown) => {
    printed.push(String(chunk));
    return true;
  }) as typeof process.stdout.write;
  try {
    await runNodeProductionRender(argv);
    results.push({ error: "", out: printed.join(""), status: 0 });
  } catch (error) {
    results.push({
      error: error instanceof Error ? error.message : String(error),
      out: printed.join(""),
      status: 1,
    });
  } finally {
    process.stdout.write = write;
  }
}
process.stderr.write(`__RENDER_ACTIONS__${JSON.stringify(results)}`);
